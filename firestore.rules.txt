rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  match /users/{uid} {
   allow read: if request.auth != null && request.auth.uid == uid;
   allow write: if false;
   match /transactions/{id} { allow read: if request.auth != null && request.auth.uid == uid; allow write: if false; }
   match /referrals/{id} { allow read: if request.auth != null && request.auth.uid == uid; allow write: if false; }
  }
  match /codes/{code} { allow read, write: if false; }
 }
}
