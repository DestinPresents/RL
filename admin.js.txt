import {initializeApp} from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-app.js';
import {getAuth,GoogleAuthProvider,signInWithPopup,onAuthStateChanged,signOut} from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-auth.js';
import {getFunctions,httpsCallable} from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-functions.js';
import {firebaseConfig} from './firebase-config.js';
const root=document.getElementById('admin-root');const safe=s=>String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));const money=n=>'₹'+Number(n||0).toLocaleString('en-IN');
let auth,fn,me,users=[],selected=null,detail=null,cursor=null,notice='',busy=false,overview=null,searchCode='',auditOpen=false;
const call=async(name,data={})=>(await httpsCallable(fn,name)(data)).data;
function render(){if(firebaseConfig.apiKey==='REPLACE_ME'){root.innerHTML='<div class="card">Set Firebase config before using admin.</div>';return}if(!me){root.innerHTML=`<div class="login"><div class="card"><div class="brand">✦ Reward<span>Lab</span></div><h2>Secure Admin Console</h2><p class="muted">Google sign-in · authorized administrators only</p>${notice?`<div class="notice">${safe(notice)}</div>`:''}<button class="btn" data-action="login">Sign in with Google</button></div></div>`;return}root.innerHTML=`<div class="admin-shell"><header class="admin-header"><div><div class="brand">✦ Reward<span>Lab</span> <span class="muted small">/ ADMIN</span></div><p class="muted small">Demo-only operations · All changes logged</p></div><div class="stack"><span class="pill">${safe(me.email)}</span><a class="btn secondary" href="index.html">User site</a><button class="btn secondary" data-action="logout">Sign out</button></div></header>${notice?`<div class="admin-alert">${safe(notice)}</div>`:''}${overview?`<div class="admin-metrics"><div class="card"><span class="muted">Total accounts</span><div class="number">${overview.totalUsers}</div></div><div class="card"><span class="muted">Pending withdrawals</span><div class="number accent">${overview.pendingWithdrawals}</div></div><div class="card"><span class="muted">Latest admin activity</span><div class="number">${overview.audit.length}</div></div></div>`:''}<div class="admin-layout"><aside class="card admin-list"><h3>Users <span class="muted small">(${users.length} loaded)</span></h3><div class="admin-toolbar"><button class="btn secondary" data-action="refresh">Refresh</button><button class="btn secondary" data-action="audit">Audit log</button></div><label class="muted small" for="searchCode">Find by 5-digit UID</label><div class="stack"><input class="field" id="searchCode" inputmode="numeric" maxlength="5" placeholder="e.g. 12345" value="${safe(searchCode)}"><button class="btn" data-action="search">Find</button></div>${users.map(u=>`<button class="user-row ${selected===u.uid?'active':''}" data-uid="${safe(u.uid)}"><strong>#${safe(u.code)} · ${safe(u.name||'Member')}</strong><div class="muted small">${safe(u.email||u.uid)}</div><div class="small">${money(u.balance)} · ${u.disabled?'Disabled':'Active'}</div></button>`).join('')||'<p class="muted">No users found.</p>'}${cursor?'<button class="btn secondary" data-action="more">Load more</button>':''}</aside><section class="admin-panel">${detail?detailHTML():auditOpen?auditHTML():`<div class="card"><h2>Manage RewardLab</h2><p class="muted">Select a user to inspect their account, edit demo credits, review referral records and process demo withdrawals.</p><div class="notice">Access is protected by Firebase custom claims. Changes cannot be made from the public client alone.</div></div>`}</section></div></div>`}
function auditHTML(){return `<div class="card"><h2>Recent admin audit activity</h2><p class="muted">Latest 20 privileged changes, recorded by the backend.</p><div class="admin-table-wrap"><table class="history"><thead><tr><th>Action</th><th>Target</th><th>Actor</th><th>When</th></tr></thead><tbody>${(overview?.audit||[]).map(x=>`<tr><td>${safe(x.action)}</td><td>${safe(x.target)}</td><td>${safe(x.actor)}</td><td>${safe(x.createdAt?.seconds?new Date(x.createdAt.seconds*1000).toLocaleString():'Pending')}</td></tr>`).join('')||'<tr><td colspan="4">No changes yet</td></tr>'}</tbody></table></div></div>`}
function detailHTML(){const u=detail.user;return `<div class="card"><div class="admin-header"><div><h2 style="margin:0">${safe(u.name||'Member')} <span class="accent">#${safe(u.code)}</span></h2><p class="muted small">${safe(u.email)} · Auth UID: ${safe(u.uid)}</p></div><span class="pill">${u.disabled?'DISABLED':'ACTIVE'}</span></div><div class="admin-stat"><div><div class="admin-label">Demo balance</div><div class="admin-value green">${money(u.balance)}</div></div><div><div class="admin-label">Total deposits</div><div class="admin-value">${money(u.totalDeposited)}</div></div><div><div class="admin-label">Referral records</div><div class="admin-value">${detail.referrals.length}</div></div></div><hr style="border-color:#303756;margin:22px 0"><h3>Edit user data</h3><p class="muted small">UID and referral attribution are immutable to preserve account integrity. Credit edits are audited.</p><div class="admin-fields"><label>Demo balance<input class="field" id="editBalance" type="number" min="0" max="100000000" step="1" value="${Number(u.balance||0)}"></label><label>Cumulative demo deposit<input class="field" id="editDeposited" type="number" min="0" max="100000000" step="1" value="${Number(u.totalDeposited||0)}"></label></div><label class="admin-label">Admin note<textarea id="editNote" class="field" rows="3" maxlength="500">${safe(u.note||'')}</textarea></label><div class="admin-toolbar"><button class="btn" data-action="save">Save changes</button><button class="btn ${u.disabled?'secondary':'danger'}" data-action="disable">${u.disabled?'Enable user':'Disable user'}</button></div></div><div class="card" style="margin-top:20px"><h3>Transaction history (latest 100)</h3><div class="admin-table-wrap"><table class="history"><thead><tr><th>Type</th><th>Amount</th><th>Status</th><th>Action</th></tr></thead><tbody>${detail.transactions.map(t=>`<tr><td>${safe(t.type)}</td><td>${money(t.amount)}</td><td class="status ${safe(t.status)}">${safe(t.status)}</td><td>${t.type==='withdrawal'&&t.status==='pending'?`<div class="stack"><button class="btn secondary" data-tx="${safe(t.id)}" data-status="success">Approve</button><button class="btn danger" data-tx="${safe(t.id)}" data-status="cancelled">Cancel</button></div>`:'—'}</td></tr>`).join('')||'<tr><td colspan="4">No records</td></tr>'}</tbody></table></div></div><div class="card" style="margin-top:20px"><h3>Referrals (latest 100)</h3><p class="muted">${detail.referrals.length?detail.referrals.map(r=>'#'+safe(r.code)).join(' · '):'No referrals'}</p></div>`}
async function run(fnAction){if(busy)return;busy=true;notice='';try{await fnAction()}catch(e){notice=e.message||String(e)}finally{busy=false;render()}}
async function loadOverview(){overview=await call('adminOverview')}
async function loadUsers(more=false){const res=await call('adminListUsers',more&&cursor?{cursor}:{});users=more?[...users,...res.users]:res.users;cursor=res.nextCursor;render()}
async function loadDetail(uid){auditOpen=false;selected=uid;detail=await call('adminGetUser',{uid});render()}
root.addEventListener('click',e=>{const t=e.target.closest('[data-action],[data-uid],[data-tx]');if(!t)return;if(t.dataset.uid)return run(()=>loadDetail(t.dataset.uid));if(t.dataset.tx)return run(async()=>{const status=t.dataset.status;if(!confirm(`Mark this DEMO withdrawal ${status}?`))return;await call('adminResolveWithdrawal',{uid:selected,transactionId:t.dataset.tx,status});notice='Withdrawal updated';await loadDetail(selected);await loadUsers()});const a=t.dataset.action;if(a==='login')return run(()=>signInWithPopup(auth,new GoogleAuthProvider()));if(a==='logout')return run(()=>signOut(auth));if(a==='refresh')return run(async()=>{await Promise.all([loadUsers(),loadOverview()]);notice='Dashboard refreshed'});if(a==='search')return run(async()=>{searchCode=document.getElementById('searchCode').value.trim();const result=await call('adminFindUser',{code:searchCode});await loadDetail(result.uid)});if(a==='audit')return run(async()=>{await loadOverview();selected=null;detail=null;auditOpen=true;notice='';render()});if(a==='more')return run(()=>loadUsers(true));if(a==='save')return run(async()=>{const balance=Number(document.getElementById('editBalance').value),totalDeposited=Number(document.getElementById('editDeposited').value),note=document.getElementById('editNote').value;if(!confirm('Save changes to this DEMO account?'))return;await call('adminEditUser',{uid:selected,patch:{balance,totalDeposited,note}});await loadDetail(selected);await loadUsers();notice='User updated successfully'});if(a==='disable')return run(async()=>{const disabled=!detail.user.disabled;if(!confirm(`${disabled?'Disable':'Enable'} this account?`))return;await call('adminSetUserDisabled',{uid:selected,disabled});await loadDetail(selected);await loadUsers();notice='Account status updated'})});
if(firebaseConfig.apiKey!=='REPLACE_ME'){const app=initializeApp(firebaseConfig);auth=getAuth(app);fn=getFunctions(app);onAuthStateChanged(auth,async u=>{me=null;users=[];detail=null;selected=null;if(u){try{const token=await u.getIdTokenResult(true);if(token.claims.admin!==true){notice='This Google account is not authorized as an administrator.';render();return}me=u;await Promise.all([loadUsers(),loadOverview()])}catch(e){notice=e.message;render()}}else render()})}render();
